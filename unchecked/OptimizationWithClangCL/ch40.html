第 40 章

CMake で同じ構成を再現する

本書は MSBuild を主軸にしてきましたが、CMake を使うプロジェクトも多いはずです。同じことを CMake で表現します。LTO には専用の仕組みがあり、PGO は自前で組むことになります。

この章のゴール

  • CMake から ClangCL を使う
  • CMake の LTO 機能を使う
  • PGO の三段階を CMake で表現する
  • MSBuild との対応を把握する

40.1 ClangCL を使う設定

二つの方法があります。

40.1.1 Visual Studio ジェネレータ ＋ ツールセット指定

cmake -S . -B build -G "Visual Studio 18 2026" -A x64 -T ClangCL

-T ClangCL が、第 3 章でプロパティページから選んだ「プラットフォーム ツールセット」に対応します。生成されるのは .sln と .vcxproj なので、実質的に第 39 章と同じ土台になります。

40.1.2 Ninja ジェネレータ ＋ コンパイラ指定

cmake -S . -B build -G Ninja ^
      -DCMAKE_C_COMPILER=clang-cl ^
      -DCMAKE_CXX_COMPILER=clang-cl ^
      -DCMAKE_LINKER=lld-link ^
      -DCMAKE_BUILD_TYPE=Release

ビルドが速く、CI との相性もよい方法です。Developer Command Prompt から実行してください（第 2 章）。環境変数が設定されていないと、標準ライブラリのヘッダが見つかりません。

どちらを選ぶか

Visual Studio でデバッグしながら開発するなら前者、CI やコマンドライン中心なら後者。本書の内容はどちらでも成立しますが、PGO の三段階をスクリプト化するなら Ninja のほうが素直です。ビルドディレクトリを分けるだけで構成を切り替えられます。

40.2 基本の CMakeLists.txt

cmake_minimum_required(VERSION 3.25)
project(ClangOptLab CXX)

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

add_executable(ClangOptLab
    src/generator.cpp
    src/tokenizer.cpp
    src/filter.cpp
    src/stats.cpp
    src/main.cpp
)
target_include_directories(ClangOptLab PRIVATE include)

# Release でもデバッグ情報を残す（第 3 章・第 14 章）
if(MSVC)
    target_compile_options(ClangOptLab PRIVATE $<$<CONFIG:Release>:/Zi>)
    target_link_options(ClangOptLab PRIVATE $<$<CONFIG:Release>:/DEBUG>)
endif()

40.3 LTO

CMake には LTO のための標準的な仕組みがあります。

include(CheckIPOSupported)
check_ipo_supported(RESULT ipo_ok OUTPUT ipo_msg)

if(ipo_ok)
    set_property(TARGET ClangOptLab PROPERTY INTERPROCEDURAL_OPTIMIZATION TRUE)
else()
    message(WARNING "LTO は利用できません: ${ipo_msg}")
endif()

実際に生成されるコマンドラインを確認すると、Clang に対しては -flto=thin が渡されます。CMake の既定は ThinLTO です（第 19 章で推奨した選択と一致します）。

$ cmake --build build -- VERBOSE=1 | grep flto
  -flto=thin

40.3.1 Full LTO にしたい場合

INTERPROCEDURAL_OPTIMIZATION は方式を選べないので、Full LTO を使うなら直接指定します。

target_compile_options(ClangOptLab PRIVATE /clang:-flto)
target_link_options(ClangOptLab PRIVATE /opt:lldlto=2)

40.3.2 ThinLTO のキャッシュ

target_link_options(ClangOptLab PRIVATE
    "/lldltocache:${CMAKE_BINARY_DIR}/ltocache"
    "/lldltocachepolicy:cache_size_bytes=2g:prune_after=168h"
)
INTERPROCEDURAL_OPTIMIZATION と手動指定を混ぜない

両方を書くと -flto=thin と -flto が両方渡り、後勝ちで意図が読めない状態になります（第 6 章）。どちらか一方に統一してください。

40.4 PGO

CMake には PGO のための標準機能がありません。オプションを自分で組み立てます。

# PGO のモードを外から指定する
set(PGO_MODE "off" CACHE STRING "off / generate / use / cs-generate")
set(PROFILE_DATA "${CMAKE_SOURCE_DIR}/app.profdata" CACHE FILEPATH "")
set(PROFILE_DIR  "${CMAKE_BINARY_DIR}/prof")

if(PGO_MODE STREQUAL "generate")
    target_compile_options(ClangOptLab PRIVATE
        "/clang:-fprofile-generate=${PROFILE_DIR}")
    target_link_directories(ClangOptLab PRIVATE "${CLANG_RT_LIB_DIR}")

elseif(PGO_MODE STREQUAL "use")
    if(NOT EXISTS "${PROFILE_DATA}")
        message(FATAL_ERROR "プロファイルがありません: ${PROFILE_DATA}")
    endif()
    target_compile_options(ClangOptLab PRIVATE
        "/clang:-fprofile-use=${PROFILE_DATA}"
        "/clang:-Werror=backend-plugin")

elseif(PGO_MODE STREQUAL "cs-generate")
    target_compile_options(ClangOptLab PRIVATE
        "/clang:-fprofile-use=${PROFILE_DATA}"
        "/clang:-fcs-profile-generate=${CMAKE_BINARY_DIR}/csprof")
endif()

リスト 40-2　PGO の切り替え

ビルドディレクトリを分けて使います。

cmake -S . -B build-instr -G Ninja -DCMAKE_CXX_COMPILER=clang-cl ^
      -DCMAKE_BUILD_TYPE=Release -DPGO_MODE=generate
cmake --build build-instr

cmake -S . -B build-pgo -G Ninja -DCMAKE_CXX_COMPILER=clang-cl ^
      -DCMAKE_BUILD_TYPE=Release -DPGO_MODE=use
cmake --build build-pgo
プロファイルランタイムのライブラリを探す

第 26 章で扱った clang_rt.profile-x86_64.lib の場所を、CMake で見つけておくと親切です。

find_path(CLANG_RT_LIB_DIR clang_rt.profile-x86_64.lib
    PATHS
        "$ENV{VCINSTALLDIR}Tools/Llvm/x64/lib/clang/*/lib/windows"
        "C:/Program Files/LLVM/lib/clang/*/lib/windows"
)
if(NOT CLANG_RT_LIB_DIR AND PGO_MODE STREQUAL "generate")
    message(FATAL_ERROR "プロファイルランタイムが見つかりません")
endif()

見つからないまま計装ビルドを試みると、第 26 章の分かりにくいリンクエラーになります。設定の段階で止めたほうが親切です。

40.5 三段階を回すスクリプト

@echo off
:: build-pgo.cmd （CMake / Ninja 版）
setlocal
set ROOT=%~dp0
set CM=cmake -S "%ROOT%" -G Ninja -DCMAKE_CXX_COMPILER=clang-cl -DCMAKE_BUILD_TYPE=Release

echo ===== [1/4] 計装ビルド =====
%CM% -B "%ROOT%build-instr" -DPGO_MODE=generate || exit /b 1
cmake --build "%ROOT%build-instr" || exit /b 1

echo ===== [2/4] トレーニング =====
if exist "%ROOT%prof" rmdir /s /q "%ROOT%prof"
mkdir "%ROOT%prof"
set LLVM_PROFILE_FILE=%ROOT%prof\run-%%p.profraw
call "%ROOT%train.cmd" "%ROOT%build-instr\ClangOptLab.exe" || exit /b 1
set LLVM_PROFILE_FILE=

echo ===== [3/4] マージ =====
llvm-profdata merge -sparse -output="%ROOT%app.profdata" "%ROOT%prof\*.profraw" || exit /b 1

echo ===== [4/4] 出荷用ビルド =====
%CM% -B "%ROOT%build-opt" -DPGO_MODE=use || exit /b 1
cmake --build "%ROOT%build-opt" || exit /b 1

echo 完了: build-opt\ClangOptLab.exe
endlocal

40.6 プリセットにまとめる

CMakePresets.json を使うと、コマンドラインが短くなります。

{
  "version": 6,
  "configurePresets": [
    {
      "name": "base",
      "hidden": true,
      "generator": "Ninja",
      "cacheVariables": {
        "CMAKE_CXX_COMPILER": "clang-cl",
        "CMAKE_BUILD_TYPE": "Release"
      }
    },
    {
      "name": "instr",
      "inherits": "base",
      "binaryDir": "${sourceDir}/build-instr",
      "cacheVariables": { "PGO_MODE": "generate" }
    },
    {
      "name": "opt",
      "inherits": "base",
      "binaryDir": "${sourceDir}/build-opt",
      "cacheVariables": { "PGO_MODE": "use", "ENABLE_LTO": "ON" }
    }
  ]
}

リスト 40-4　CMakePresets.json

cmake --preset instr && cmake --build build-instr
cmake --preset opt   && cmake --build build-opt

40.8 CMake ならではの注意点

キャッシュが残る

PGO_MODE を変えて同じビルドディレクトリを使い回すと、キャッシュ変数が残って意図しない組み合わせになることがあります。モードごとにビルドディレクトリを分けるのが確実です。

ジェネレータ式のクォート

/clang:-fprofile-use=... のようにコロンを含むオプションは、ジェネレータ式（$<...>）の中で使うと構文が壊れます。SHELL: プレフィックスや、素直な if() による分岐を使ってください。

パスの空白

C:\Program Files\LLVM のように空白を含むパスは、必ずクォートしてください。リスト 40-2 で "..." を付けているのはそのためです。

40.9 ハンズオン

  1. リスト 40-1 の CMakeLists.txt を作り、Ninja でビルドする
  2. checksum が変わらないことを確認する
  3. INTERPROCEDURAL_OPTIMIZATION を有効にし、VERBOSE=1 で -flto=thin が渡ることを確認する
  4. リスト 40-2 の PGO_MODE を追加する
  5. build-instr と build-opt を作り、三段階を通す
  6. プロファイルを消して PGO_MODE=use を試し、設定段階でエラーになることを確認する

40.10 まとめ

  • ClangCL は -T ClangCL（VS ジェネレータ）か CMAKE_CXX_COMPILER=clang-cl（Ninja）で使う
  • Ninja を使う場合は Developer Command Prompt から
  • LTO は INTERPROCEDURAL_OPTIMIZATION で有効になり、Clang では -flto=thin が渡る
  • Full LTO やキャッシュは手動でオプションを指定する。両方を混ぜない
  • PGO の標準機能は無い。キャッシュ変数（PGO_MODE）とビルドディレクトリの分離で表現する
  • プロファイルランタイムの場所は find_path で探し、無ければ設定段階で止める
  • CMakePresets.json でコマンドラインを短くできる
